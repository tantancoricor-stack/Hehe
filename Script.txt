-- ✈️ FLY FUNCTION
local function FlyTo(targetPos)
    if isFlying then return end
    isFlying = true
    
    local B = Instance.new("BodyVelocity")
    B.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    B.Velocity = Vector3.new(0, 0, 0)
    B.Parent = HRP

    local distance = (HRP.Position - targetPos).Magnitude
    local time = distance / FLY_SPEED
    
    local direction = (targetPos - HRP.Position).Unit * FLY_SPEED
    B.Velocity = direction
    
    game:GetService("RunService").Heartbeat:Wait()
    repeat
        task.wait(0.01)
    until (HRP.Position - targetPos).Magnitude < 10
    
    B:Destroy()
    isFlying = false
end

-- 🥚 FIND NEAREST EGG + PREDICTION
local function FindNearestEgg()
    local nearest = nil
    local dist = math.huge
    
    -- ✅ I-adjust mo dito ang path kung saan nandun ang mga eggs sa laro
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") and v.Name:match("Egg") or v.Name:match("egg") then
            -- Simple prediction logic base on name/color
            local eggData = {
                Part = v,
                Position = v.Position,
                Prediction = "???"
            }
            
            -- Guess kung anong nasa loob base sa hitsura/pangalan
            if v.Name:find("Common") then eggData.Prediction = "Common"
            elseif v.Name:find("Uncommon") then eggData.Prediction = "Uncommon"
            elseif v.Name:find("Rare") then eggData.Prediction = "Rare"
            elseif v.Name:find("Epic") then eggData.Prediction = "Epic"
            elseif v.Name:find("Legendary") then eggData.Prediction = "Legendary"
            elseif v.Name:find("Mythic") then eggData.Prediction = "Mythic"
            elseif v.Name:find("Secret") then eggData.Prediction = "SECRET ⭐"
            end
            
            local d = (HRP.Position - v.Position).Magnitude
            if d < dist then
                dist = d
                nearest = eggData
            end
        end
    end
    return nearest
end

-- ✅ GET EGG LOGIC
GetBtn.MouseButton1Click:Connect(function()
    if not selectedPet then
        warn("[TEC HUB] Pumili muna ng pet sa list!")
        return
    end
    
    autoLoop = true
    GetBtn.Text = "🔄 WORKING..."
    GetBtn.BackgroundColor3 = Color3.fromRGB(220, 170, 0)
    
    task.spawn(function()
        while autoLoop do
            -- Step 1: Hanapin ang pinakamalapit na egg
            local egg = FindNearestEgg()
            if not egg then
                task.wait(1)
                continue
            end
            
            print("[TEC HUB] Egg found! Predicted: " .. egg.Prediction)
            
            -- Step 2: Teleport/Fly papunta sa egg
            FlyTo(egg.Position)
            task.wait(0.3)
            
            -- Step 3: Kunin ang egg (touch/interact)
            HRP.CFrame = CFrame.new(egg.Position) * CFrame.new(0, -2, 0)
            task.wait(0.2)
            if egg.Part then
                firetouchinterest(HRP, egg.Part, 0)
                firetouchinterest(HRP, egg.Part, 1)
            end
            task.wait(0.3)
            
            -- Step 4: Fly pabalik sa base ✅ I-update mo ang base position dito
            local BASE_POSITION = Vector3.new(0, 10, 0) -- ⚠️ PALITAN MO ITO NG TUNAY NA BASE POSITION
            FlyTo(BASE_POSITION)
            task.wait(0.5)
            
            print("[TEC HUB] Done! Back to base.")
            task.wait(0.5)
        end
    end)
end)

-- Drag Panel
local dragToggle, dragStart, startPos
local function UpdateDrag(input)
    local delta = input.Position - dragStart
    Panel.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end
Panel.InputBegan:Connect(function(v)
    if v.UserInputType == Enum.UserInputType.MouseButton1 then
                dragToggle = true
                dragStart = v.Position
                startPos = Panel.Position
                end
end)
UIS.InputChanged:Connect(function(v) if dragToggle and v.UserInputType == Enum.UserInputType.MouseMovement then UpdateDrag(v) end end)
UIS.InputEnded:Connect(function() dragToggle = false end)
